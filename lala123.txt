import getpass # Para obtener el nombre de usuario de la PC
import os

# Define aquí la ruta de la carpeta compartida del banco
# Asegúrate de usar 'r' antes de las comillas para rutas de Windows
RUTA_COMPARTIDA = r"\\servidor_scotia\riesgos\monitor_status"




def exportar_estado_jefe(self, automatico=False):
        try:
            # 1. Obtener usuario de red
            usuario = getpass.getuser()
            nombre_archivo = f"{usuario}_status.xlsx"
            ruta_final = os.path.join(RUTA_COMPARTIDA, nombre_archivo)

            # 2. Verificar si la carpeta compartida es accesible
            if not os.path.exists(RUTA_COMPARTIDA):
                if not automatico:
                    messagebox.showerror("Error de Ruta", "No se puede acceder a la carpeta compartida.")
                return

            # 3. Crear el Excel de Status
            wb = openpyxl.Workbook()
            ws = wb.active
            ws.title = "Status Actual"
            
            # Encabezados
            ws.append(["Proceso", "Estado", "Avance", "Ítem / Correo", "Hora Recibido", "Validación"])
            
            # Recorrer los resultados actuales (basándonos en la lógica de sincronización)
            # Aquí usamos el último 'resultado' obtenido por el monitor
            if hasattr(self, 'ultimo_resultado') and self.ultimo_resultado:
                grupos = self.ultimo_resultado.get("grupos", {})
                
                for nombre_g, datos in grupos.items():
                    estado = "COMPLETO" if datos["cumplidos"] == datos["total"] else "EN PROCESO"
                    avance = f"{datos['cumplidos']}/{datos['total']}"
                    
                    # Añadir filas de recibidos
                    for rec in datos["recibidos"]:
                        val_str = "Conforme" if rec.get("es_valido", True) else "OBSERVADO"
                        ws.append([nombre_g, estado, avance, rec["nombre"], rec["hora"], val_str])
                    
                    # Añadir filas de faltantes
                    for falt in datos["faltantes"]:
                        ws.append([nombre_g, estado, avance, falt, "PENDIENTE", "PENDIENTE"])

            wb.save(ruta_final)
            
            if not automatico:
                messagebox.showinfo("Reporte Generado", f"Tu estado se ha actualizado en:\n{ruta_final}")
                
        except Exception as e:
            print(f"Error en exportar_estado_jefe: {e}")

    # Función para el ciclo automático de 1 hora
    def _ciclo_reporte_automatico(self):
        # Primero actualizamos los datos de Outlook
        self.sincronizar_correos()
        # Luego exportamos al compartido de forma silenciosa
        self.after(5000, lambda: self.exportar_estado_jefe(automatico=True))
        # Programamos la siguiente ejecución en 1 hora (3,600,000 ms)
        self.after(3600000, self._ciclo_reporte_automatico)
