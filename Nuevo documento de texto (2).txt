def inicializar_excel():
    """Crea el archivo Excel con las hojas necesarias si no existe."""
    if os.path.exists(EXCEL_FILE):
        return
    wb = Workbook()
 
    # Hoja 1: Reglas Diarias
    ws1 = wb.active
    ws1.title = "Reglas Diarias"
    ws1.append(["nombre", "remitente", "asunto", "frecuencia", "activa"])
    ws1.column_dimensions["A"].width = 28
    ws1.column_dimensions["B"].width = 35
    ws1.column_dimensions["C"].width = 35
    ws1.column_dimensions["D"].width = 18
    ws1.column_dimensions["E"].width = 10
 
    # Hoja 2: Reglas Calendario
    ws2 = wb.create_sheet("Reglas Calendario")
    ws2.append(["nombre", "remitente", "asunto", "grupo", "fecha_inicio", "fecha_fin", "activa"])
    ws2.column_dimensions["A"].width = 28
    ws2.column_dimensions["B"].width = 35
    ws2.column_dimensions["C"].width = 35
    ws2.column_dimensions["D"].width = 20
    ws2.column_dimensions["E"].width = 14
    ws2.column_dimensions["F"].width = 14
    ws2.column_dimensions["G"].width = 10
 
    wb.save(EXCEL_FILE)
 
 
def leer_reglas_diarias():
    inicializar_excel()
    wb = openpyxl.load_workbook(EXCEL_FILE)
    ws = wb["Reglas Diarias"]
    reglas = []
    for row in ws.iter_rows(min_row=2, values_only=True):
        if row[0]:
            reglas.append({
                "nombre":     str(row[0]).strip(),
                "remitente":  str(row[1]).strip() if row[1] else "",
                "asunto":     str(row[2]).strip() if row[2] else "",
                "frecuencia": str(row[3]).strip() if row[3] else "Hoy",
                "activa":     str(row[4]).strip().lower() not in ("false", "no", "0") if row[4] else True,
            })
    return reglas

def eliminar_regla_excel(nombre_regla: str):
    inicializar_excel()
    wb = openpyxl.load_workbook(EXCEL_FILE)
    for sheet_name in ["Reglas Diarias", "Reglas Calendario"]:
        ws = wb[sheet_name]
        rows_to_delete = []
        for i, row in enumerate(ws.iter_rows(min_row=2, values_only=True), start=2):
            if row[0] and str(row[0]).strip() == nombre_regla:
                rows_to_delete.append(i)
        for idx in reversed(rows_to_delete):
            ws.delete_rows(idx)
    wb.save(EXCEL_FILE)
 
 
def agregar_ignorado_excel(nombre_regla: str, entry_id: str):
    """Agrega un entry_id a la columna 'ignorados' (columna F en Diarias, H en Calendario)."""
    inicializar_excel()
    wb = openpyxl.load_workbook(EXCEL_FILE)
    for sheet_name, ignorados_col in [("Reglas Diarias", 6), ("Reglas Calendario", 8)]:
        ws = wb[sheet_name]
        for row in ws.iter_rows(min_row=2):
            if row[0].value and str(row[0].value).strip() == nombre_regla:
                existing = row[ignorados_col - 1].value or ""
                ids = [x for x in existing.split("|") if x]
                if entry_id not in ids:
                    ids.append(entry_id)
                row[ignorados_col - 1].value = "|".join(ids)
                break
    wb.save(EXCEL_FILE)
 
 
def leer_ignorados_excel(nombre_regla: str) -> list:
    inicializar_excel()
    wb = openpyxl.load_workbook(EXCEL_FILE)
    for sheet_name, ignorados_col in [("Reglas Diarias", 6), ("Reglas Calendario", 8)]:
        ws = wb[sheet_name]
        for row in ws.iter_rows(min_row=2, values_only=True):
            if row[0] and str(row[0]).strip() == nombre_regla:
                val = row[ignorados_col - 1] or ""
                return [x for x in str(val).split("|") if x]
    return []

    def _iniciar_sync_thread(self):
        if self._sync_running:
            return
        self.btn_sync.configure(text="⏳  Actualizando…", state="disabled", fg_color=SCOTIA_BORDER)
        self.lbl_status.configure(text="Conectando con Outlook…")
        self._sync_running = True
        t = threading.Thread(target=self._sync_worker, daemon=True)
        t.start()
 
    def _sync_worker(self):
        """Ejecuta la lógica pesada en segundo plano."""
        try:
            resultado = self._obtener_resultados_outlook()
        except Exception as e:
            resultado = {"error": str(e)}
        # Regresar a hilo principal para actualizar UI
        self.after(0, lambda: self._sync_finalizado(resultado))
 
    def _obtener_resultados_outlook(self) -> dict:
        """Conecta a Outlook y cruza contra reglas. Corre en hilo secundario."""
        reglas_diarias    = leer_reglas_diarias()
        reglas_calendario = leer_reglas_calendario()
        hoy = datetime.datetime.now().date()
 
        # Filtrar reglas activas
        reglas = [r for r in reglas_diarias if r.get("activa", True)]
 
        # Agregar reglas de calendario que aplican hoy
        for r in reglas_calendario:
            if not r.get("activa", True):
                continue
            try:
                fi = datetime.datetime.strptime(r["fecha_inicio"], "%d/%m/%Y").date()
                ff = datetime.datetime.strptime(r["fecha_fin"],    "%d/%m/%Y").date()
                if fi <= hoy <= ff:
                    reglas.append(r)
            except Exception:
                pass
 
        if not reglas:
            return {"grupos": {}, "simples": [], "sin_reglas": True}
 
        # Conectar a Outlook
        outlook   = win32com.client.Dispatch("Outlook.Application").GetNamespace("MAPI")
        bandeja   = outlook.GetDefaultFolder(6)
        mensajes  = bandeja.Items
        mensajes.Sort("[ReceivedTime]", True)
 
        max_correos = min(2000, len(mensajes))
 
        resultados_simples = []
        resultados_grupos  = {}
 
        for regla in reglas:
            cumplida    = False
            hora_str    = ""
            correo_id   = ""
            dias_margin = 0
            grupo       = regla.get("grupo", "")
            ignorados   = leer_ignorados_excel(regla["nombre"])
 
            # Calcular ventana de tiempo
            if "fecha_inicio" in regla:
                fecha_limite = datetime.datetime.strptime(regla["fecha_inicio"], "%d/%m/%Y").date()
            else:
                freq = regla.get("frecuencia", "Hoy")
                if freq == "Últimos 3 días":
                    dias_margin = 3
                elif freq == "Últimos 7 días":
                    dias_margin = 7
                fecha_limite = hoy - datetime.timedelta(days=dias_margin)
 
            asunto_regla   = regla["asunto"].lower()
            remitente_regla = regla["remitente"].lower()
            palabras_clave  = asunto_regla.split()
 
            for i in range(max_correos):
                try:
                    msg = mensajes.Item(i + 1)
                    if getattr(msg, "Class", 0) != 43:
                        continue
 
                    if msg.SenderEmailType == "EX":
                        remitente = msg.Sender.GetExchangeUser().PrimarySmtpAddress
                    else:
                        remitente = msg.SenderEmailAddress
                    if not remitente:
                        remitente = msg.SenderName
 
                    asunto     = msg.Subject if msg.Subject else ""
                    fecha_msg  = msg.ReceivedTime.date()
                    entry_id   = msg.EntryID
                    cuerpo     = msg.Body if msg.Body else ""
 
                    if fecha_msg < fecha_limite:
                        break
 
                    if entry_id in ignorados:
                        continue
 
                    coincide_remitente = remitente_regla in remitente.lower()
                    coincide_asunto    = all(p in asunto.lower() for p in palabras_clave)
 
                    if coincide_remitente and coincide_asunto:
                        hora_str  = str(msg.ReceivedTime)[:16]
                        cumplida  = True
                        correo_id = entry_id
                        es_valido = self.evaluar_contenido_conforme(cuerpo)
                        break
                except Exception:
                    continue
 
            # Calcular días restantes si es regla de calendario
            dias_restantes = None
            if "fecha_fin" in regla:
                try:
                    ff_date = datetime.datetime.strptime(regla["fecha_fin"], "%d/%m/%Y").date()
                    dias_restantes = (ff_date - hoy).days
                except Exception:
                    pass
 
            valido = es_valido if cumplida and 'es_valido' in dir() else True
            id_c   = correo_id if cumplida else ""
 
            if not grupo:
                resultados_simples.append({
                    "nombre":         regla["nombre"],
                    "cumplida":       cumplida,
                    "hora":           hora_str,
                    "dias_restantes": dias_restantes,
                    "es_valido":      valido,
                    "correo_id":      id_c,
                })
            else:
                if grupo not in resultados_grupos:
                    resultados_grupos[grupo] = {
                        "total": 0, "cumplidos": 0,
                        "faltantes": [], "recibidos": [],
                        "dias_restantes": dias_restantes,
                        "fecha_inicio": regla.get("fecha_inicio", ""),
                        "fecha_fin":    regla.get("fecha_fin", ""),
                    }
                resultados_grupos[grupo]["total"] += 1
                if cumplida:
                    resultados_grupos[grupo]["cumplidos"] += 1
                    resultados_grupos[grupo]["recibidos"].append({
                        "nombre": regla["nombre"], "hora": hora_str,
                        "es_valido": valido, "correo_id": id_c,
                    })
                else:
                    resultados_grupos[grupo]["faltantes"].append(regla["nombre"])
 
        return {
            "grupos":    resultados_grupos,
            "simples":   resultados_simples,
            "sin_reglas": False,
            "error":     None,
        }
 
    def _sync_finalizado(self, resultado: dict):
        """Llamado en el hilo principal al terminar la sincronización."""
        self._sync_running = False
        self.btn_sync.configure(text="🔄  Actualizar", state="normal", fg_color=SCOTIA_YELLOW)
 
        if resultado.get("error"):
            self.lbl_status.configure(
                text=f"⚠  Error: {resultado['error']}", text_color="#FF6B6B"
            )
            return
 
        if resultado.get("sin_reglas"):
            self._limpiar_paneles()
            ctk.CTkLabel(
                self.left_panel,
                text="No hay reglas activas para hoy.",
                text_color=SCOTIA_SUBTEXT, font=("Trebuchet MS", 12)
            ).pack(pady=20)
            self._actualizar_hora()
            return
 
        self._dibujar_resultados(resultado["grupos"], resultado["simples"])
        self._actualizar_hora()
 
    def _limpiar_paneles(self):
        for w in self.left_panel.winfo_children():
            w.destroy()
        for w in self.right_panel.winfo_children():
            w.destroy()
 
    def _actualizar_hora(self):
        hora = datetime.datetime.now().strftime("%H:%M:%S")
        self.lbl_status.configure(
            text=f"Última actualización: {hora}", text_color=SCOTIA_SUBTEXT
        )
        self.lbl_hora.configure(text=hora)
 
    # ----------------------------------------------------------
    #  DIBUJAR RESULTADOS
    # ----------------------------------------------------------
    def _dibujar_resultados(self, grupos: dict, simples: list):
        self._limpiar_paneles()
 
        # — Grupos —
        for nombre_grupo, datos in grupos.items():
            self._dibujar_grupo(nombre_grupo, datos)
 
        # — Reglas simples faltantes —
        simples_faltantes = [r for r in simples if not r["cumplida"]]
        simples_recibidos = [r for r in simples if r["cumplida"]]
 
        if simples_faltantes:
            make_section_label(self.left_panel, "▸ TAREAS INDIVIDUALES").pack(
                anchor="w", padx=10, pady=(10, 4)
            )
            for res in simples_faltantes:
                self._dibujar_faltante_simple(res)
 
        if simples_recibidos:
            make_section_label(self.right_panel, "▸ TAREAS INDIVIDUALES").pack(
                anchor="w", padx=10, pady=(10, 4)
            )
            for res in simples_recibidos:
                self._dibujar_recibido_simple(res)
 
    def _dibujar_grupo(self, nombre_grupo: str, datos: dict):
        dias_r = datos["dias_restantes"]
        if dias_r is None:
            texto_dias = ""
        elif dias_r > 0:
            texto_dias = f"  ({dias_r} días restantes)"
        elif dias_r == 0:
            texto_dias = "  (¡Vence HOY!)"
        else:
            texto_dias = f"  (Venció hace {abs(dias_r)} días)"
 
        if datos["cumplidos"] == datos["total"]:
            # Panel verde (completado)
            card = ctk.CTkFrame(
                self.right_panel, fg_color=SCOTIA_CARD2,
                corner_radius=8, border_width=1, border_color=SCOTIA_GREEN_DIM
            )
            card.pack(anchor="w", fill="x", padx=4, pady=4)
 
            header = ctk.CTkFrame(card, fg_color=SCOTIA_GREEN_DIM, corner_radius=0, height=30)
            header.pack(fill="x")
            header.pack_propagate(False)
            ctk.CTkLabel(

    def iniciar_temporizador(self, ventana):
        destinatario  = self.entry_destinatario.get()
        frecuencia_str = self.combo_horas.get()
        try:
            horas = int(frecuencia_str.split(" ")[0])
        except Exception:
            horas = 1
        milisegundos = horas * 60 * 60 * 1000
 
        self.automatizacion_activa = True
        self.btn_alerta.configure(
            text=f"🔴  Alertas Activas ({horas}h) — Clic para detener",
            fg_color="#1B3A1B", hover_color="#0A1F0A",
            command=self.desactivar_automatizacion
        )
        ventana.destroy()
        self.ciclo_alerta(destinatario, milisegundos)
 
    def ciclo_alerta(self, destinatario: str, ms_espera: int):
        if not self.automatizacion_activa:
            return
        self.enviar_correo_alerta(destinatario)
        self.timer_id = self.after(ms_espera, lambda: self.ciclo_alerta(destinatario, ms_espera))
 
    def desactivar_automatizacion(self):
        self.automatizacion_activa = False
        if self.timer_id is not None:
            self.after_cancel(self.timer_id)
            self.timer_id = None
        self.btn_alerta.configure(
            text="⏰  Programar Alertas",
            fg_color=SCOTIA_RED_DARK, hover_color="#6B0609",
            command=self.abrir_config_alerta
        )
 
    # ----------------------------------------------------------
    #  ENVIAR CORREO DE ALERTA
    # ----------------------------------------------------------
    def enviar_correo_alerta(self, destinatario: str):
        self._iniciar_sync_thread()
 
        faltantes = []
        for widget in self.left_panel.winfo_children():
            if isinstance(widget, ctk.CTkCheckBox):
                faltantes.append(widget.cget("text"))
            elif isinstance(widget, ctk.CTkFrame):
                for sub in widget.winfo_children():
                    if isinstance(sub, ctk.CTkCheckBox):
                        faltantes.append(sub.cget("text"))
 
        if not faltantes:
            return
 
        try:
            outlook = win32com.client.Dispatch("Outlook.Application")
            correo  = outlook.CreateItem(0)
            correo.To      = destinatario
            correo.Subject = "Recordatorio: Reportes y Conformidades Pendientes de Envío"
 
            lista_html = ""
            for item in faltantes:
                nombre_limpio = item.replace("Falta: ", "")
                lista_html   += f'<li style="margin-bottom:5px"><b>{nombre_limpio}</b></li>'
 
            correo.HTMLBody = f"""
<html><body style="font-family:'Segoe UI',Arial,sans-serif;color:#262626;line-height:1.6">
  <h3 style="color:#ED0006;border-bottom:1px solid #ED0006;padding-bottom:5px">
    Aviso Automático: Monitoreo de Reportes
  </h3>
  <p>Estimado equipo,</p>
  <p>Por medio de la presente, les recordamos que al corte de la revisión actual,
     el sistema registra los siguientes reportes/conformidades como
     <b>pendientes de envío</b>:</p>
  <ul style="color:#ED0006"><span style="color:#262626">{lista_html}</span></ul>
  <p>Agradeceremos contactar a los encargados correspondientes para asegurar
     que la información llegue a la brevedad posible.</p>
  <br>
  <p style="font-size:12px;color:#555555">
    Atentamente,<br><br>
    <b>Área de Riesgos de Mercado</b><br>
    <b style="color:#ED0006">Scotiabank Perú</b><br>
    <i>Este es un mensaje generado automáticamente por el Monitor de Reportes.</i>
  </p>
</body></html>"""
            correo.Send()
        except Exception as e:
            print(f"Error al enviar correo de alerta: {e}")

    # ----------------------------------------------------------
    #  EVALUADOR DE CONTENIDO
    # ----------------------------------------------------------
    def evaluar_contenido_conforme(self, cuerpo_correo: str) -> bool:
        if not cuerpo_correo:
            return True
        texto_limpio = cuerpo_correo
        for sep in ["\r\nDe:", "\nDe:", "\r\nFrom:", "\nFrom:"]:
            if sep in texto_limpio:
                texto_limpio = texto_limpio.split(sep)[0]
        texto = texto_limpio.lower()
        alertas = [
            "error", "equivoca", "revisar", "no cuadra", "observación",
            "observacion", "rechaz", "falta", "corrige", "incorrecto",
            "pendiente", "diferencia", "ajustar", "volver", "modifiquemos",
            "comentario", "cambiemos", "mejora", "no me coincide",
            "no coincide", "cambia", "cambie",
        ]
        for palabra in alertas:
            if palabra in texto:
                return False
        return True
